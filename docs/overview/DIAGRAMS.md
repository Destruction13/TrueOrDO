     │                                              │ Broadcast to all
     │                                              │ clients in room
     │                                              │

│                        1. СОЗДАНИЕ КОМНАТЫ                       │
└──────────────────────────────────────────────────────────────────┘
                              │
                              ▼
                    socket.emit("alias:create")
                              │

                              ▼

                              ▼
                  ┌───────────────────────┐

                  └───────────┬───────────┘

                              ▼

                  │  Генерация колоды     │
                  │  слов                 │
                  └───────────┬───────────┘
                              │
                              ▼
                  ┌───────────────────────┐
                  │  Выбор первой команды │
                  │  и объясняющего       │
                  └───────────┬───────────┘
                              │
                              ▼
                  ┌───────────────────────┐
                  │  Запуск таймера       │
                  └───────────┬───────────┘
                              │
                              ▼
      socket.emit("alias:word:next", { word, ... })

                              ▼
          io.to(roomCode).emit("alias:room:state", {...})

┌──────────────────────────────────────────────────────────────────┐
│                       5. ИГРОВОЙ ПРОЦЕСС                         │
